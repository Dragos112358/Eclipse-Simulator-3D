# -----------------------------------------------------------------------------
# CALCUL NUMERIC
# -----------------------------------------------------------------------------
# NumPy: toate calculele vectorizate - vectori ECEF/ITRS ai orasenlor, cautarea
# ternara pentru contactele orasenesti, interpolarea traseului umbrei (np.interp),
# produse vectoriale pentru latimea benzii de totalitate etc.
numpy>=1.24


# -----------------------------------------------------------------------------
# ASTRONOMIE / EFEMERIDE
# -----------------------------------------------------------------------------
# Skyfield: nucleul astronomic al serverului. Foloseste:
#   - Loader / timescale (ts.utc, ts.tt_jd, Delta T din IERS finals2000A.all)
#   - wgs84 (pozitia observatorului pe elipsoid)
#   - almanac (faze lunare, rasarit/apus)
#   - framelib.itrs (coordonate fixe pe Pamant), positionlib.Geocentric
#   - constants.AU_KM, errors.EphemerisRangeError
# Descarca automat (langa executabil) fisierele:
#   de441.bsp (planete + Luna), finals2000A.all (IERS),
#   jup365.bsp, sat441.bsp, nep097.bsp (sateliti Jupiter/Saturn/Neptun).
# NOTA: de441.bsp este MARE (~3 GB); jup365 / sat441 / nep097 au sute de MB.
skyfield>=1.46

# Skyfield depinde si de "jplephem" (citirea fisierelor .bsp) si "sgp4".
# Se instaleaza automat, dar il listam explicit pentru claritate / reproductibilitate.
jplephem>=2.18
sgp4>=2.20


# -----------------------------------------------------------------------------
# GEOLOCATIE (reverse geocoding: lat/lon -> nume localitate)
# -----------------------------------------------------------------------------
# geopy: foloseste serviciul Nominatim (OpenStreetMap) in get_location_name().
# Necesita conexiune la internet; la esec, serverul revine pe "GPS (lat, lon)".
geopy>=2.4


# -----------------------------------------------------------------------------
# RETEA NEURALA - CORECTIA DE LIMB LUNAR (relief LOLA)
# -----------------------------------------------------------------------------
# PyTorch: ruleaza modelul LunarAltitudeNet (hash-grid encoding + MLP) incarcat din
# "lunar_altitude_contact_bands_fp16_finetuned.pt". Se foloseste DOAR pe CPU
# (device="cpu"), deci versiunea CPU-only este suficienta si mult mai mica.
#
# Pentru varianta CPU-only (recomandat, ~200 MB in loc de ~2.5 GB), instaleaza manual:
#     pip install torch --index-url https://download.pytorch.org/whl/cpu
# Linia de mai jos functioneaza oricum (descarca varianta standard).
torch>=2.1

# OPTIONAL - impachetare ca executabil (.exe)

# Decomenteaza daca vrei sa compilezi serverul:
# pyinstaller>=6.0
